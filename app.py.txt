import os
import streamlit as st
import chromadb
from google import genai

st.set_page_config(page_title="SAP Basis RAG Copilot", page_icon="🛠️")
st.title("🛠️ SAP Basis Incident Diagnostic Copilot")
st.caption("AI-powered Root Cause & SAP Note Assistant using RAG + Gemini")

# Sidebar for API Key
api_key = st.sidebar.text_input("Enter Gemini API Key", type="password")
st.sidebar.markdown("[Get Free Gemini API Key](https://google.com)")

@st.cache_resource
def setup_vector_db():
    client = chromadb.Client()
    collection = client.get_or_create_collection(name="sap_notes")
    
    if collection.count() == 0:
        with open("knowledge_base.txt", "r") as f:
            content = f.read()
        
        # Split knowledge base by SAP Note blocks
        chunks = [chunk.strip() for chunk in content.split("\n\n") if chunk.strip()]
        ids = [f"note_{i}" for i in range(len(chunks))]
        collection.add(documents=chunks, ids=ids)
    return collection

collection = setup_vector_db()

# User Input
log_input = st.text_area(
    "Paste ST22 Dump / SM21 Log Error here:",
    placeholder="Example: Runtime Errors TSV_TNEW_PAGE_ALLOC_FAILED occurred during batch job Z_FI_EXTRACT...",
    height=130
)

if st.button("Diagnose Issue"):
    if not api_key:
        st.error("Please enter your Gemini API Key in the sidebar first!")
    elif not log_input.strip():
        st.warning("Please paste an SAP error log to analyze.")
    else:
        with st.spinner("Searching SAP Knowledge Base & Generating Root Cause..."):
            # 1. Retrieve relevant SAP Note from ChromaDB
            results = collection.query(query_texts=[log_input], n_results=1)
            retrieved_context = results["documents"][0][0]
            
            # 2. Generate Diagnostic Response using Gemini
            ai_client = genai.Client(api_key=api_key)
            prompt = f"""
            You are an expert Senior SAP Basis & Infrastructure AI Copilot.
            Analyze the user's SAP error log using the retrieved SAP Note context below.
            Provide a structured response with:
            1. **Matched SAP Note & Category**
            2. **Probable Root Cause**
            3. **Step-by-Step Basis Resolution (with T-Codes)**
            
            Retrieved Knowledge Base:
            {retrieved_context}
            
            User Error Log:
            {log_input}
            """
            
            response = ai_client.models.generate_content(
                model="gemini-2.5-flash",
                contents=prompt
            )
            
            st.success("Diagnosis Complete!")
            st.markdown(response.text)
            
            with st.expander("View Retrieved SAP Note Chunk (RAG Context)"):
                st.code(retrieved_context)
